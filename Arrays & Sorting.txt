ARRAYS
Program 1: Sum of Array Elements

#include <iostream>
using namespace std;

int main()
{
    int arr[5], sum = 0;

    for(int i = 0; i < 5; i++)
    {
        cout << "Enter number: ";
        cin >> arr[i];

        sum = sum + arr[i];
    }

    cout << "Sum = " << sum;

    return 0;
}

Program 2: Maximum and Minimum

#include <iostream>
using namespace std;

int main()
{
    int arr[5];

    for(int i = 0; i < 5; i++)
    {
        cout << "Enter number: ";
        cin >> arr[i];
    }

    int max = arr[0], min = arr[0];

    for(int i = 1; i < 5; i++)
    {
        if(arr[i] > max)
            max = arr[i];

        if(arr[i] < min)
            min = arr[i];
    }

    cout << "Maximum = " << max << endl;
    cout << "Minimum = " << min;

    return 0;
}

SORTING 

Program 1: Ascending Sorting

#include <iostream>
using namespace std;

int main()
{
    int arr[5], temp;

    for(int i = 0; i < 5; i++)
    {
        cout << "Enter number: ";
        cin >> arr[i];
    }

    for(int i = 0; i < 5; i++)
    {
        for(int j = 0; j < 4; j++)
        {
            if(arr[j] > arr[j + 1])
            {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }

    cout << "Ascending Order: ";

    for(int i = 0; i < 5; i++)
    {
        cout << arr[i] << " ";
    }

    return 0;
}

Program 3: Bubble Sort

#include <iostream>
using namespace std;

int main()
{
    int arr[5], temp;

    for(int i = 0; i < 5; i++)
    {
        cout << "Enter number: ";
        cin >> arr[i];
    }

    for(int i = 0; i < 4; i++)
    {
        for(int j = 0; j < 4 - i; j++)
        {
            if(arr[j] > arr[j + 1])
            {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }

    cout << "Sorted Array: ";

    for(int i = 0; i < 5; i++)
    {
        cout << arr[i] << " ";
    }

    return 0;
}
