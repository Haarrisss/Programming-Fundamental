Program 1: Star Triangle

#include <iostream>
using namespace std;

int main()
{
    for(int i = 1; i <= 5; i++)
    {
        for(int j = 1; j <= i; j++)
        {
            cout << "* ";
        }

        cout << endl;
    }

    return 0;
}

Program 3: Student Marks (5 Students, 3 Subjects)

#include <iostream>
using namespace std;

int main()
{
    int marks, total;

    for(int i = 1; i <= 5; i++)
    {
        total = 0;

        cout << "Student " << i << endl;

        for(int j = 1; j <= 3; j++)
        {
            cout << "Enter marks of subject " << j << ": ";
            cin >> marks;

            total = total + marks;
        }

        cout << "Total marks = " << total << endl;
    }

    return 0;
}

Program 4: Attendance System (5 Students, 7 Days)

#include <iostream>
using namespace std;

int main()
{
    int attendance, total;

    for(int i = 1; i <= 5; i++)
    {
        total = 0;

        cout << "Student " << i << endl;

        for(int j = 1; j <= 7; j++)
        {
            cout << "Day " << j << " (1=Present, 0=Absent): ";
            cin >> attendance;

            total = total + attendance;
        }

        cout << "Total Present Days = " << total << endl;
    }

    return 0;
}

Program 5: Multiplication Tables (1 to 5)

#include <iostream>
using namespace std;

int main()
{
    for(int i = 1; i <= 5; i++)
    {
        cout << "Table of " << i << endl;

        for(int j = 1; j <= 10; j++)
        {
            cout << i << " x " << j << " = " << i * j << endl;
        }

        cout << endl;
    }

    return 0;
}
